    ```js
    /**
     * Generate the 1200×630 Open Graph share-card image (Phase 10 / SEO-02).
     * Run: node scripts/generate-og-image.mjs
     *
     * Output: public/og-image-v1.png — versioned filename per D-10 / SEO-08
     * so future image swaps (og-image-v2.png) defeat Facebook's OG cache.
     *
     * Palette is the warm-earth source-of-truth from src/index.css @theme block.
     * Sharp + librsvg renders Georgia + system-ui fonts without registration
     * (proven by scripts/generate-icons.mjs on Windows/macOS/Linux).
     */
    import sharp from 'sharp';
    import { fileURLToPath } from 'node:url';
    import { dirname, resolve } from 'node:path';

    const __dirname = dirname(fileURLToPath(import.meta.url));
    const outputPath = resolve(__dirname, '..', 'public', 'og-image-v1.png');

    // Palette — copied from src/index.css @theme block (warm earth)
    const WARM_EARTH_BG = '#f4f1eb';   // --color-bg
    const SAGE = '#5c6b56';            // --color-sage
    const TEXT_PRIMARY = '#3d4a38';    // --color-text-primary

    function createOgSvg() {
      return `<svg xmlns="http://www.w3.org/2000/svg" width="1200" height="630" viewBox="0 0 1200 630">
      <rect width="1200" height="630" fill="${WARM_EARTH_BG}"/>
      <!-- Decorative ring glyph (singing-bowl / chime reference) -->
      <circle cx="240" cy="315" r="80" fill="none" stroke="${SAGE}" stroke-width="14"/>
      <circle cx="240" cy="315" r="120" fill="none" stroke="${SAGE}" stroke-width="6" opacity="0.35"/>
      <!-- Wordmark — Georgia italic per project icon convention -->
      <text
        x="380" y="305"
        font-family="Georgia, 'Times New Roman', serif"
        font-size="140"
        font-style="italic"
        font-weight="400"
        fill="${SAGE}"
        dominant-baseline="middle"
      >Soundly</text>
      <!-- Tagline — system-ui sans, deep forest -->
      <text
        x="380" y="395"
        font-family="system-ui, -apple-system, 'Helvetica Neue', Arial, sans-serif"
        font-size="42"
        font-weight="400"
        fill="${TEXT_PRIMARY}"
      >a calmer alarm</text>
    </svg>`;
    }

    async function generate() {
      await sharp(Buffer.from(createOgSvg()))
        .png({ compressionLevel: 9 })
        .toFile(outputPath);
      console.log(`Created ${outputPath}`);
    }

    generate().catch((err) => {
      console.error(err);
      process.exit(1);
    });
    ```

    - "a calmer alarm" tagline in system-ui sans, deep forest — calmer secondary tone

    - No stock-photo people, no jarring colors (D-08 LOCKED constraint)

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| scripts/generate-og-image.mjs → public/og-image-v1.png | Generator is a build-time Node script; output is a static asset. Script runs only when explicitly invoked (`npm run og-image`), NOT on every build. |
| public/og-image-v1.png → social-media scrapers | Anyone with the share URL can request the image. It is public by design. |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-10-04-01 | Tampering | OG image content | mitigate | D-09 user-approval checkpoint blocks commit until visual review. Versioned filename (D-10) means any post-deploy swap forces a new URL → Facebook cache miss → fresh scrape. |
| T-10-04-02 | Information Disclosure | OG image as untrusted shared-link target | accept | OG image is intentionally public — no PII. Whoever has the deploy URL can fetch the image; that is the point of OG. |
| T-10-04-03 | Denial of Service | malicious large PNG | mitigate | Automated verifier asserts size ≤ 200 KB SEO-02 cap. sharp's PNG output for vector content is reliably 5-50 KB; a malicious script could embed a 200 MB raster but Task 1's `createOgSvg()` produces pure-vector SVG with bounded content. |
| T-10-04-04 | Spoofing | substituted OG image at same URL | mitigate | HTTPS-only deploy (GitHub Pages enforces). Versioned filename (`og-image-v1.png`) means swapping content goes through a new URL (`og-image-v2.png`), so Facebook can't be tricked into showing v1 content after a v2 deploy. |
| T-10-04-05 | Elevation of Privilege | sharp-via-SVG SVG-injection | mitigate | The SVG is hand-authored inside the script; no user input flows into the SVG string. sharp + librsvg do NOT execute embedded `<script>` tags in SVG. No injection vector. |
| T-10-04-06 | Repudiation | image content edits | mitigate | D-09 user-approval checkpoint creates a human-in-the-loop audit step; git history records the approved PNG hash. |
</threat_model>
